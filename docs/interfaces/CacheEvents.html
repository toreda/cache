    Interface CacheEvents<ItemT>

    User-supplied observability callbacks fired by Cache. Every callback is optional and costs nothing when unset. Callbacks fire post-commit (state + stats already updated), synchronously, each wrapped in try/catch — a throwing callback is logged and never corrupts cache state.

    Single-item removals fire the specific event (onItemExpire / onItemEvict) first, then the general onItemRemove. Bulk clear() / reset() fire only onClear (and onReset) — never per-item events.

    interface CacheEvents<ItemT> {
        onAddReject?: (item: ItemT, reason: CacheRejectReason) => void;
        onCapacityDecrease?: (oldMax: number, newMax: number) => void;
        onCapacityIncrease?: (oldMax: number, newMax: number) => void;
        onClear?: (count: number) => void;
        onItemAdd?: (item: ItemT, id: string) => void;
        onItemEvict?: (item: ItemT, id: string) => void;
        onItemExpire?: (item: ItemT, id: string, timeAdded: Time) => void;
        onItemHit?: (item: ItemT, id: string, source: "get" | "touch") => void;
        onItemMiss?: (id: string) => void;
        onItemRemove?: (item: ItemT, id: string, reason: CacheRemoveReason) => void;
        onPrune?: (removed: number) => void;
        onReset?: () => void;
    }
